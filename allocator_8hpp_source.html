216 struct size_index {
217 static constexpr size_t NUM_BUCKETS = 10;
218 free_block* buckets[NUM_BUCKETS] = {nullptr};
219
220 static size_t get_bucket(uint64_t size) {
221 if (size <= 256) return 0;
222 if (size <= 512) return 1;
223 if (size <= 1024) return 2;
224 if (size <= 2048) return 3;
225 if (size <= 4096) return 4;
226 if (size <= 8192) return 5;
227 if (size <= 16384) return 6;
228 if (size <= 32768) return 7;
229 if (size <= 65536) return 8;
230 return 9;
231 }
232
233 void insert(free_block *block) {
234 size_t bucket = get_bucket(block->size);
235 block->next_in_bucket = buckets[bucket];
236 block->prev_in_bucket = nullptr;
237 if (buckets[bucket]) {
238 buckets[bucket]->prev_in_bucket = block;
239 }
240 buckets[bucket] = block;
241 }
242
243 void remove(free_block *block) {
244 size_t bucket = get_bucket(block->size);
245 if (block->prev_in_bucket) {
246 block->prev_in_bucket->next_in_bucket = block->next_in_bucket;
247 } else {
248 buckets[bucket] = block->next_in_bucket;
249 }
250 if (block->next_in_bucket) {
251 block->next_in_bucket->prev_in_bucket = block->prev_in_bucket;
252 }
253 block->next_in_bucket = nullptr;
254 block->prev_in_bucket = nullptr;
255 }
256
257 void clear() {
258 for (size_t i = 0; i < NUM_BUCKETS; ++i) {
259 buckets[i] = nullptr;
260 }
261 }
262
263 free_block *find_best_fit(uint64_t size) const {
264 size_t start_bucket = get_bucket(size);
265 free_block* best = nullptr;
266
267 for (size_t i = start_bucket; i < NUM_BUCKETS; ++i) {
268 for (free_block* block = buckets[i]; block; block = block->next_in_bucket) {
269 if (block->size >= size) {
270 if (!best || block->size < best->size) {
271 best = block;
272 if (block->size == size) return best; // Exact fit
273 }
274 }
275 }
276 // If found in current bucket, it's the best fit
277 if (best && i == start_bucket) return best;
278 }
279 return best;
280 }
281
282 free_block *find_worst_fit(uint64_t size) const {
283 // Start from largest bucket
284 for (int i = NUM_BUCKETS - 1; i >= 0; --i) {
285 free_block* largest = nullptr;
286 for (free_block* block = buckets[i]; block; block = block->next_in_bucket) {
287 if (block->size >= size) {
288 if (!largest || block->size > largest->size) {
289 largest = block;
290 }
291 }
292 }
293 if (largest) return largest;
294 }
295 return nullptr;
296 }
