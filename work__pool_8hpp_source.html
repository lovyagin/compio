15// Runs the iterations of a loop on several threads; the calling thread takes part.
16// Workers start at the first loop that has more than one iteration.
17class work_pool {
18public:
19 explicit work_pool(unsigned threads) : threads_(threads == 0 ? 1 : threads) {}
20
21 ~work_pool() {
22 {
23 std::lock_guard<std::mutex> lock(mutex_);
24 stop_ = true;
25 }
26 start_.notify_all();
27 for (auto &worker : workers_) worker.join();
28 }
29
30 work_pool(const work_pool &) = delete;
31 work_pool &operator=(const work_pool &) = delete;
32
33 unsigned threads() const { return threads_; }
34
35 void run(size_t count, const std::function<void(size_t)> &body) {
36 if (threads_ == 1 || count < 2) {
37 for (size_t i = 0; i < count; ++i) body(i);
38 return;
39 }
40 std::lock_guard<std::mutex> serial(run_mutex_);
41 {
42 std::lock_guard<std::mutex> lock(mutex_);
43 if (workers_.empty()) {
44 for (unsigned i = 1; i < threads_; ++i) workers_.emplace_back([this] { work(); });
45 }
46 body_ = &body;
47 count_ = count;
48 next_.store(0);
49 busy_ = static_cast<unsigned>(workers_.size());
50 ++generation_;
51 }
52 start_.notify_all();
53 take(body, count);
54 std::unique_lock<std::mutex> lock(mutex_);
55 done_.wait(lock, [this] { return busy_ == 0; });
56 body_ = nullptr;
57 }
58
59private:
60 const unsigned threads_;
61 std::vector<std::thread> workers_;
62 std::mutex run_mutex_;
63 std::mutex mutex_;
64 std::condition_variable start_;
65 std::condition_variable done_;
66 const std::function<void(size_t)> *body_ = nullptr;
67 size_t count_ = 0;
68 std::atomic<size_t> next_{0};
69 unsigned busy_ = 0;
70 uint64_t generation_ = 0;
71 bool stop_ = false;
72
73 void take(const std::function<void(size_t)> &body, size_t count) {
74 for (size_t i = next_.fetch_add(1); i < count; i = next_.fetch_add(1)) body(i);
75 }
76
77 void work() {
78 uint64_t seen = 0;
79 std::unique_lock<std::mutex> lock(mutex_);
80 while (true) {
81 start_.wait(lock, [&] { return stop_ || generation_ != seen; });
82 if (stop_) return;
83 seen = generation_;
84 const auto *body = body_;
85 const size_t count = count_;
86 lock.unlock();
87 take(*body, count);
88 lock.lock();
89 if (--busy_ == 0) done_.notify_one();
90 }
91 }
92};
